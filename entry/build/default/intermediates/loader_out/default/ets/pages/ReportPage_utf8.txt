// 适配级别颜色映射条目
class LevelColorEntry {
  key: string = '';
  value: string = '';
}

const LEVEL_COLORS: LevelColorEntry[] = [
  { key: '适合', value: COLORS.SUCCESS },
  { key: '少量可以', value: COLORS.WARNING },
  { key: '今天谨慎', value: COLORS.DANGER },
  { key: '建议避免', value: COLORS.DANGER }
];

const LEVEL_BG_COLORS: LevelColorEntry[] = [
  { key: '适合', value: COLORS.ACCENT_GREEN_LIGHT },
  { key: '少量可以', value: COLORS.SECONDARY_LIGHT },
  { key: '今天谨慎', value: COLORS.BG_CARD_HIGHLIGHT },
  { key: '建议避免', value: COLORS.BG_SECTION }
];

// 成员角色对应的头像emoji
class RoleEmojiEntry {
  role: string = '';
  emoji: string = '';
}

const ROLE_EMOJIS: RoleEmojiEntry[] = [
  { role: '爸爸', emoji: '👨' },

  { role: '孩子', emoji: '👶' },
  { role: '我', emoji: '🧑' }
];

@Entry
@Component
struct ReportPage {
  @State foodLabel: FoodLabel = new FoodLabel()
  @State familyMembers: FamilyProfile[] = []
  @State recommendations: Recommendation[] = []
  @State selectedMemberId: string = ''
  @State selectedRecommendation: Recommendation | null = null
  @State childProtection: ChildProtection | null = null
  @State credibilityResult: CredibilityResult = new CredibilityResult()
  @State healthScoreResult: HealthScoreResult | null = null
  @State alternativeFoods: AlternativeFood[] = []
  @State showAlternativeFoods: boolean = false
  @State unifiedScore: UnifiedScoreResult | null = null
  @State dailyBudget: DailyBudget | null = null
  @State nextMealAdvice: string = ''
  @State manufacturerName: string = ''
  @State scNumber: string = ''
  @State entrustInfo: string = ''
  @State dataCompleteness: number = 0

  private appState: AppState = AppState.getInstance()
  private foodId: string = ''
  private historySaved: boolean = false

  async aboutToAppear() {
    const params = router.getParams() as Record<string, string>
    if (params) {
      this.foodId = params.foodId || ''
      this.selectedMemberId = params.memberId || ''
    }
    await this.loadReportData()
  }

  build() {
    Scroll() {
      Column() {
        // 顶部导航
        Row() {
          Text('←')
            .fontSize(FONT_SIZE.LARGE)
            .fontColor(COLORS.PRIMARY)
            .onClick(() => { router.back() })
          Text('家庭适配报告')
            .fontSize(FONT_SIZE.TITLE_LG)
            .fontWeight(FONT_WEIGHT.BOLD)
            .fontColor(COLORS.TEXT_PRIMARY)
            .margin({ left: SPACING.MD })
        }
        .width('100%')
        .padding({ left: SPACING.LG, right: SPACING.LG, top: SPACING.LG, bottom: SPACING.LG })

        // 食品信息卡片
        this.FoodInfoCard()

        // 成员选择器
        if (this.familyMembers.length > 0) {
          this.MemberSelector()
        }

        // 家庭对比卡片
        if (this.recommendations.length > 0) {
          this.FamilyCompareCard()
        }

        // 选中成员详情
        if (this.selectedRecommendation !== null) {
          this.SelectedMemberDetail()
        }

        // 替代品推荐
        if (this.showAlternativeFoods && this.alternativeFoods.length > 0) {
          this.AlternativeFoodSection()
        }

        // 儿童正餐保护
        if (this.childProtection !== null && this.childProtection.hasAnyAdvice()) {
          this.ChildProtectionSection()
        }

        // 综合评分
        if (this.unifiedScore !== null) {
          this.UnifiedScoreSection()
        }

        // 下一餐建议
        if (this.nextMealAdvice.length > 0) {
          this.NextMealAdviceSection()
        }

        // 今日剩余额度
        if (this.dailyBudget !== null) {
          this.BudgetOverviewSection()
        }

        // 操作按钮
        this.ActionButtons()

        // 底部声明
        this.FooterDisclaimer()
      }
      .width('100%')
      .alignItems(HorizontalAlign.Center)
    }
    .width('100%')
    .height('100%')
    .backgroundColor(COLORS.BG_PAGE)
  }

  // ==================== 食品信息卡片 ====================
  @Builder
  FoodInfoCard() {
    Column() {
      // 食品名称
      Text(this.foodLabel.foodName.length > 0 ? this.foodLabel.foodName : '待补充食品名称')
        .fontSize(FONT_SIZE.TITLE)
        .fontWeight(FONT_WEIGHT.BOLD)
        .fontColor(COLORS.TEXT_PRIMARY)
        .margin({ bottom: SPACING.SM })

      // 营养成分网格
      Row() {
        this.NutriChip('钠', `${this.foodLabel.nutrition.sodium}mg`, this.foodLabel.isHighSodium())
        this.NutriChip('糖', `${this.foodLabel.nutrition.sugar}g`, this.foodLabel.isHighSugar())
        this.NutriChip('热量', `${this.foodLabel.nutrition.calories}kcal`, this.foodLabel.isHighCalorie())
        this.NutriChip('脂肪', `${this.foodLabel.nutrition.fat}g`, this.foodLabel.isHighFat())
      }
      .width('100%')
      .justifyContent(FlexAlign.SpaceBetween)
      .margin({ bottom: SPACING.SM })

      // 条码信息
      if (this.foodLabel.barcode.length > 0) {
        Text(`条码：${this.foodLabel.barcode}`)
          .fontSize(FONT_SIZE.SMALL)
          .fontColor(COLORS.TEXT_HINT)
          .margin({ bottom: SPACING.XS })
      }
    }
    .width('92%')
    .padding(SPACING.LG)
    .backgroundColor(COLORS.BG_CARD)
    .borderRadius(RADIUS.LG)
    .alignItems(HorizontalAlign.Start)
    .margin({ bottom: SPACING.SM })
    .shadow({ radius: 8, color: COLORS.SHADOW_LIGHT, offsetX: 0, offsetY: 2 })
  }

  // ==================== 营养成分标签 ====================
  @Builder
  NutriChip(label: string, value: string, isHigh: boolean) {
    Column() {
      Text(label)
        .fontSize(FONT_SIZE.TINY)
        .fontColor(COLORS.TEXT_HINT)
      Text(value)
        .fontSize(FONT_SIZE.SMALL)
        .fontWeight(FONT_WEIGHT.MEDIUM)
        .fontColor(isHigh ? COLORS.DANGER : COLORS.TEXT_PRIMARY)
        .margin({ top: SPACING.XS })
    }
    .padding({ left: SPACING.SM, right: SPACING.SM, top: SPACING.SM, bottom: SPACING.SM })
    .backgroundColor(isHigh ? COLORS.DANGER_LIGHT : COLORS.BG_SECTION)
    .borderRadius(RADIUS.SM)
    .alignItems(HorizontalAlign.Center)
  }

  // ==================== 成员选择器 ====================
  @Builder
  MemberSelector() {
    Column() {
      Text('查看谁的结论')
        .fontSize(FONT_SIZE.SMALL)
        .fontColor(COLORS.TEXT_HINT)
        .margin({ bottom: SPACING.SM })

      Row() {
        ForEach(this.familyMembers, (member: FamilyProfile) => {
          Column() {
            Text(this.getMemberEmoji(member))
              .fontSize(FONT_SIZE.XLARGE)
            Text(member.nickname)
              .fontSize(FONT_SIZE.TINY)
              .fontColor(this.selectedMemberId === member.memberId ? COLORS.TEXT_WHITE : COLORS.TEXT_PRIMARY)
              .margin({ top: SPACING.XS })
          }
          .width(64)
          .height(64)
          .justifyContent(FlexAlign.Center)
          .alignItems(HorizontalAlign.Center)
          .borderRadius(RADIUS.FULL)
          .backgroundColor(this.selectedMemberId === member.memberId ? COLORS.PRIMARY : COLORS.BG_SECTION)
          .margin({ right: SPACING.SM })
          .onClick(() => { this.selectMember(member.memberId) })
        }, (member: FamilyProfile) => member.memberId)
      }
      .width('100%')
      .justifyContent(FlexAlign.Center)
    }
    .width('92%')

    .backgroundColor(COLORS.BG_CARD)
    .borderRadius(RADIUS.LG)
    .alignItems(HorizontalAlign.Start)
    .margin({ bottom: SPACING.SM })
  }

          .margin({ bottom: SPACING.SM })

    }
    .width('92%')
    .padding(SPACING.LG)
    .backgroundColor(COLORS.BG_CARD)
    .borderRadius(RADIUS.LG)
    .alignItems(HorizontalAlign.Start)
    .margin({ bottom: SPACING.SM })
    .shadow({ radius: 8, color: COLORS.SHADOW_LIGHT, offsetX: 0, offsetY: 2 })
  }

            Text(this.getMemberName(this.selectedMemberId))

          .layoutWeight(1)
        }
        .width('100%')
        .margin({ bottom: SPACING.MD })

        // 建议级别大卡片

          Column() {
            Text(this.selectedRecommendation.level)
              .fontSize(FONT_SIZE.XLARGE)

              .fontColor(COLORS.TEXT_WHITE)
            Text(this.getAmountText(this.selectedRecommendation.maxAmount))
              .fontSize(FONT_SIZE.SMALL)
              .fontColor(COLORS.TEXT_WHITE)

          }
          .width('100%')
          .padding(SPACING.LG)
          .backgroundColor(getLevelColorValue(this.selectedRecommendation.levelColor))
          .borderRadius(RADIUS.LG)
          .alignItems(HorizontalAlign.Center)
        }
        .width('100%')
        .margin({ bottom: SPACING.MD })

        // 原因列表
        if (this.selectedRecommendation.reasons.length > 0) {
          Text('📋 原因')
            .fontSize(FONT_SIZE.BODY)
            .fontWeight(FONT_WEIGHT.MEDIUM)
            .fontColor(COLORS.TEXT_SECONDARY)
            .margin({ bottom: SPACING.SM })
          ForEach(this.selectedRecommendation.reasons, (reason: string) => {
            Text(`• ${reason}`)
              .fontSize(FONT_SIZE.BODY)
              .fontColor(COLORS.TEXT_PRIMARY)
              .margin({ bottom: SPACING.XS })
          }, (reason: string, index: number) => `reason_${index}`)
        }

        // 提醒列表
        if (this.selectedRecommendation.reminders.length > 0) {
          Text('💡 提醒')
            .fontSize(FONT_SIZE.BODY)
            .fontWeight(FONT_WEIGHT.MEDIUM)
            .fontColor(COLORS.TEXT_SECONDARY)
            .margin({ top: SPACING.MD, bottom: SPACING.SM })
          ForEach(this.selectedRecommendation.reminders, (reminder: string) => {
            Text(`• ${reminder}`)
              .fontSize(FONT_SIZE.BODY)
              .fontColor(COLORS.TEXT_HINT)
              .margin({ bottom: SPACING.XS })
          }, (reminder: string, index: number) => `reminder_${index}`)
        }
      }
      .width('92%')
      .padding(SPACING.LG)
      .backgroundColor(COLORS.BG_CARD)
      .borderRadius(RADIUS.LG)
      .alignItems(HorizontalAlign.Start)
      .margin({ bottom: SPACING.SM })
      .shadow({ radius: 8, color: COLORS.SHADOW_LIGHT, offsetX: 0, offsetY: 2 })
    }
  }

  @Builder
  BudgetProgressRow(label: string, remaining: number, budget: number, color: string) {
    Column() {
      Row() {
        Text(label)
          .fontSize(FONT_SIZE.SMALL)
          .fontColor(COLORS.TEXT_PRIMARY)
        Blank()
        Text(`${Math.round(Math.max(0, budget - remaining))} / ${Math.round(budget)}`)
          .fontSize(FONT_SIZE.SMALL)
          .fontColor(COLORS.TEXT_HINT)
      }
      .width('100%')
      .margin({ bottom: SPACING.XS })

      Row() {
        Row()
          .width(`${budget > 0 ? Math.min(100, Math.round(Math.max(0, budget - remaining) / budget * 100)) : 0}%`)
          .height(8)
          .backgroundColor(this.getBudgetBarColor(Math.max(0, budget - remaining), budget, color))
          .borderRadius(RADIUS.SM)
      }
      .width('100%')
      .height(8)

      .borderRadius(RADIUS.SM)

  // ==================== 工具方法 ====================
  private getMemberEmoji(member: FamilyProfile): string {
    for (const entry of ROLE_EMOJIS) {
      if (entry.role === member.nickname) {
        return entry.emoji
      }
    }
    return '🧑'
  }

  private getMemberEmojiById(memberId: string): string {
    const member = this.familyMembers.find(item => item.memberId === memberId)
    if (member !== undefined) {
      return this.getMemberEmoji(member)
    }
    return '🧑'
  }

  private getMemberName(memberId: string): string {
    const member = this.familyMembers.find(item => item.memberId === memberId)
    return member !== undefined ? member.nickname : '未知成员'
  }

  private getHealthGoalText(memberId: string): string {
    const member = this.familyMembers.find(item => item.memberId === memberId)
    if (member !== undefined) {
      const goals = member.healthGoals
      if (goals.length > 0) {
        return goals.join('、')
      }
    }
    return '日常饮食'
  }

  private getAmountText(maxAmount: number): string {
    if (maxAmount > 0) {
      return `建议上限：${maxAmount}g`
    }
    return '建议避免食用'
  }

  private getLevelColor(level: string): string {
    for (const entry of LEVEL_COLORS) {
      if (entry.key === level) {
        return entry.value
      }
    }
    return COLORS.TEXT_HINT
  }

  private getLevelBgColor(level: string): string {
    for (const entry of LEVEL_BG_COLORS) {
      if (entry.key === level) {
        return entry.value
      }
    }
    return COLORS.BG_SECTION
  }

  private getBudgetStatusLabel(budget: DailyBudget): string {
    const totalUsage = (budget.sodiumUsagePercent() + budget.sugarUsagePercent() +
      budget.calorieUsagePercent() + budget.fatUsagePercent()) / 4
    if (totalUsage > 80) {
      return '⚠️ 今日额度紧张'
    } else if (totalUsage > 50) {
      return '⚡ 注意控制'
    }
    return '✅ 额度充足'
  }

  private getBudgetStatusColor(budget: DailyBudget): string {
    const totalUsage = (budget.sodiumUsagePercent() + budget.sugarUsagePercent() +
      budget.calorieUsagePercent() + budget.fatUsagePercent()) / 4
    if (totalUsage > 80) {
      return COLORS.DANGER
    } else if (totalUsage > 50) {
      return COLORS.WARNING
    }
    return COLORS.SUCCESS
  }

  private getBudgetStatusBgColor(budget: DailyBudget): string {
    const totalUsage = (budget.sodiumUsagePercent() + budget.sugarUsagePercent() +
      budget.calorieUsagePercent() + budget.fatUsagePercent()) / 4
    if (totalUsage > 80) {
      return COLORS.DANGER_LIGHT
    } else if (totalUsage > 50) {
      return COLORS.WARNING_LIGHT
    }
    return COLORS.ACCENT_GREEN_LIGHT
  }

  private getBudgetBarColor(consumed: number, budget: number, color: string): string {
    const ratio = budget > 0 ? consumed / budget : 0
    if (ratio > 0.8) {
      return COLORS.DANGER
    } else if (ratio > 0.5) {
      return COLORS.WARNING
    }
    return color
  }

  private selectMember(memberId: string) {
    this.selectedMemberId = memberId
    const index = this.familyMembers.findIndex(item => item.memberId === memberId)
    if (index >= 0 && index < this.recommendations.length) {
      this.selectedRecommendation = this.recommendations[index]
    }
  }

  private async loadReportData() {
    try {
      // 加载食品标签
      if (this.foodId.length > 0) {
        const label = await this.appState.findFoodById(this.foodId)
        if (label !== null) {
          this.foodLabel = label
        }
      }

      // 加载家庭成员
      const members = await this.appState.getAllMembers()
      this.familyMembers = members

      // 为每个成员计算适配建议
      if (this.familyMembers.length > 0) {
        const recs: Recommendation[] = []
        for (const member of this.familyMembers) {
          const rec = this.appState.calculateRecommendation(member.memberId, this.foodLabel)
          recs.push(rec)
        }
        this.recommendations = recs

        // 默认选中第一个成员
        if (this.selectedMemberId.length === 0) {
          this.selectedMemberId = this.familyMembers[0].memberId
        }
        this.selectMember(this.selectedMemberId)
      }

      // 计算健康评分（使用选中成员）
      if (this.selectedMemberId.length > 0) {
        this.healthScoreResult = this.appState.calculateHealthScore(this.selectedMemberId, this.foodLabel)
      }

      // 获取替代品推荐
      if (this.selectedMemberId.length > 0) {
        this.alternativeFoods = this.appState.recommendAlternatives(this.selectedMemberId, this.foodLabel)
      }
      this.showAlternativeFoods = this.alternativeFoods.length > 0

      // 获取今日预算
      if (this.selectedMemberId.length > 0) {
        this.dailyBudget = this.appState.getDailyBudget(this.selectedMemberId)
      }

      // 获取下一餐建议
      if (this.selectedMemberId.length > 0) {
        this.nextMealAdvice = this.appState.getNextMealAdvice(this.foodLabel, this.selectedMemberId)
      }
    } catch (error) {
      console.error('ReportPage loadReportData error:', error)
    }
  }
